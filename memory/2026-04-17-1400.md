/**
 * Classify a user request and return the ordered model chain.
 *
 * @param {string} request – Raw user message.
 * @returns {{
 *   intent: 'CODING'|'STUDY'|'GENERAL',
 *   models: string[]   // OpenRouter IDs in fallback order
 * }}
 */
function routeRequest(request) {
  const lower = request.toLowerCase();

  // ----- CODING detection (highest priority) -----
  const codingIndicators = [
    /```[\s\S]*?```/,                                   // code fences
    /\b(python|js|javascript|java|c\+\+|c#|go|rust|php|ruby|bash|sh|sql)\b/,
    /git\s+(clone|push|pull|commit|branch|merge|rebase)/,
    /debug|stack trace|error|exception/,
    /api\s+design|endpoint|rest|graphql/,
    /performance\s+optim(ization|ise)/,
    /refactor|architecture|design pattern/
  ];
  const isCoding = codingIndicators.some(r => r.test(lower));

  // ----- STUDY detection (next priority) -----
  const studyIndicators = [
    /explain|how (does|do|to) /,
    /teach me|learn|study|concept/,
    /essay|report|paper|summary/,
    /pdf|document|analysis/,
    /exam|sat|quiz|test|homework/,
    /math|science|history|biology|physics|chemistry/
  ];
  const isStudy = studyIndicators.some(r => r.test(lower));

  // ----- GENERAL detection (fallback) -----
  const generalIndicators = [
    /business|idea|plan|productivity|advice|opinion|casual|todo|schedule/
  ];
  const isGeneral = generalIndicators.some(r => r.test(lower));

  // ----- Resolve ambiguity per spec (coding > study > general) -----
  let intent;
  if (isCoding) intent = 'CODING';
  else if (isStudy) intent = 'STUDY';

  else intent = 'GENERAL';   // safest default
